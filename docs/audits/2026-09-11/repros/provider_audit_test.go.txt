func TestAuditGeminiStreamAccumulates(t *testing.T) {
	s := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		w.Header().Set("Content-Type", "text/event-stream")
		fmt.Fprint(w, "data: {\"candidates\":[{\"content\":{\"role\":\"model\",\"parts\":[{\"text\":\"Hello \"}]}}]}\n\n")
		fmt.Fprint(w, "data: {\"candidates\":[{\"content\":{\"role\":\"model\",\"parts\":[{\"text\":\"world\"}]},\"finishReason\":\"STOP\"}],\"usageMetadata\":{\"promptTokenCount\":5,\"candidatesTokenCount\":2}}\n\n")
	}))
	defer s.Close()
	c, err := NewGeminiClientWithBaseURL(context.Background(), "key", "gemini-test", s.URL)
	if err != nil {
		t.Fatal(err)
	}
	ch, err := c.CreateMessageStream(context.Background(), &Request{Messages: []Message{NewUserMessage("hi")}})
	if err != nil {
		t.Fatal(err)
	}
	var delta string
	var final *Response
	for e := range ch {
		if e.Error != nil {
			t.Fatal(e.Error)
		}
		delta += e.Text
		if e.Type == EventMessageStop {
			final = e.Response
		}
	}
	if final == nil || final.TextContent() != delta {
		t.Fatalf("deltas=%q final=%+v", delta, final)
	}
}
func TestAuditGeminiToolMetadata(t *testing.T) {
	original := &genai.GenerateContentResponse{Candidates: []*genai.Candidate{{Content: &genai.Content{Role: "model", Parts: []*genai.Part{{ThoughtSignature: []byte("opaque"), FunctionCall: &genai.FunctionCall{ID: "call_1", Name: "read", Args: map[string]any{"p": "a"}}}}}}}}
	blocks := convertGeminiResponse(original, "gemini-test").Content
	replay := convertMessage(Message{Role: RoleAssistant, Blocks: blocks}).Parts[0]
	if replay.FunctionCall.ID != "call_1" {
		t.Errorf("function call ID lost: %+v", replay.FunctionCall)
	}
	if string(replay.ThoughtSignature) != "opaque" {
		t.Errorf("thought signature lost: %q", replay.ThoughtSignature)
	}
	result := convertMessage(Message{Role: RoleUser, Blocks: []ContentBlock{{Type: ContentTypeToolResult, Name: "read", ToolUseID: "call_1", Text: "ok"}}}).Parts[0]
	if result.FunctionResponse.ID != "call_1" {
		t.Errorf("result ID lost: %+v", result.FunctionResponse)
	}
}

func TestAuditRetryInitialHTTPFailure(t *testing.T) {
	var count atomic.Int32
	s := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		count.Add(1)
		w.Header().Set("Content-Type", "application/json")
		w.WriteHeader(503)
		fmt.Fprint(w, `{"error":{"message":"unavailable","type":"server_error"}}`)
	}))
	defer s.Close()
	c := &OpenAIClient{client: openai.NewClient(option.WithAPIKey("key"), option.WithBaseURL(s.URL), option.WithMaxRetries(0)), model: "model"}
	retry := NewRetryClient(c, &RetryConfig{MaxRetries: 1})
	ch, err := retry.CreateMessageStream(context.Background(), &Request{Messages: []Message{NewUserMessage("hi")}})
	if err == nil {
		for range ch {
		}
	}
	if count.Load() != 2 {
		t.Fatalf("requests=%d expected 2 on initial 503; synchronous err=%v", count.Load(), err)
	}
}
func TestAuditCancellationFullStream(t *testing.T) {
	s := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		w.Header().Set("Content-Type", "text/event-stream")
		for i := 0; i < 250; i++ {
			fmt.Fprint(w, "data: {\"type\":\"response.output_text.delta\",\"delta\":\"x\"}\n\n")
		}
	}))
	defer s.Close()
	c := NewOpenAIClientWithBaseURL("key", "model", s.URL)
	ctx, cancel := context.WithCancel(context.Background())
	defer cancel()
	ch, err := c.CreateMessageStream(ctx, &Request{Messages: []Message{NewUserMessage("hi")}})
	if err != nil {
		t.Fatal(err)
	}
	deadline := time.Now().Add(time.Second)
	for len(ch) < 100 && time.Now().Before(deadline) {
		time.Sleep(time.Millisecond)
	}
	cancel()
	time.Sleep(30 * time.Millisecond)
	b := make([]byte, 1<<20)
	n := runtime.Stack(b, true)
	leaked := strings.Contains(string(b[:n]), "(*OpenAIClient).CreateMessageStream.func1")
	for range ch {
	} // cleanup producer after collecting evidence
	if leaked {
		t.Fatal("producer remains blocked on full channel after cancellation")
	}
}
